-- 1. Load the library (replace the URL with the actual one if needed)
local Lib = loadstring(game:HttpGet("https://raw.githubusercontent.com/Mamaksimaaa/Roblox-script/refs/heads/main/Minecraft%20library/Minecraftlibrary.lua"))()

-- 2. Create a window with Overworld theme (can be Nether or End)
local Window = Lib:CreateWindow("CFrame Speed", {Theme = "Overworld"})

-- 3. Tab for speed control
local SpeedTab = Window:AddTab("Speed")

-- Control variables
local speedEnabled = false
local speedMultiplier = 5  -- default 5x
local speedConnection = nil -- RunService.Heartbeat connection
local player = game.Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()
local humanoid = character:WaitForChild("Humanoid")
local rootPart = character:WaitForChild("HumanoidRootPart")

-- Function to start/stop CFrame speed boosting
local function updateSpeedLoop()
    if speedConnection then
        speedConnection:Disconnect()
        speedConnection = nil
    end

    if speedEnabled then
        speedConnection = game:GetService("RunService").Heartbeat:Connect(function(dt)
            -- Check that the character and its parts are alive
            if not character or not character.Parent or not humanoid or humanoid.Health <= 0 then
                return
            end
            if not rootPart or not rootPart.Parent then
                return
            end

            -- Movement direction (WASD)
            local moveDirection = humanoid.MoveDirection
            if moveDirection.Magnitude > 0 then
                -- Base walking speed (usually 16 studs/sec)
                local baseSpeed = humanoid.WalkSpeed
                local distance = baseSpeed * speedMultiplier * dt
                local offset = moveDirection * distance

                -- Move the character directly via CFrame
                rootPart.CFrame = rootPart.CFrame + offset
            end
        end)
    end
end

-- Update character references on respawn
local function onCharacterAdded(newCharacter)
    character = newCharacter
    humanoid = character:WaitForChild("Humanoid")
    rootPart = character:WaitForChild("HumanoidRootPart")
    -- Restart the loop if speed was enabled
    if speedEnabled then
        updateSpeedLoop()
    end
end
player.CharacterAdded:Connect(onCharacterAdded)

-- Handle enable/disable of speed
local function setSpeedEnabled(state)
    speedEnabled = state
    updateSpeedLoop()
    Window:Notify("Speed", state and "CFrame Speed: ON" or "CFrame Speed: OFF", 2)
end

-- Create UI elements

-- Toggle for enabling/disabling (returns an object you can later modify)
local Toggle = SpeedTab:AddToggle("Enable CFrame Speed", false, function(state)
    setSpeedEnabled(state)
end)

-- Slider for speed multiplier (returns an object with Set/Get)
local Slider = SpeedTab:AddSlider("Speed Multiplier", 1, 50, speedMultiplier, function(value)
    speedMultiplier = value
    Window:Notify("Speed", "Multiplier set to " .. value .. "x", 1)
    -- If speed is enabled, update the loop with the new multiplier (not strictly necessary, but clean)
    if speedEnabled then
        updateSpeedLoop()
    end
end)

-- Button to reset speed (no state, just callback)
SpeedTab:AddButton("Reset Speed", function()
    Slider:Set(5)          -- change slider value via its Set method
    speedMultiplier = 5
    if speedEnabled then
        updateSpeedLoop()
    end
    Window:Notify("Speed", "Speed reset to 5x", 2)
end)

-- Separator and label (for visual grouping)
SpeedTab:AddSeparator("Info")
SpeedTab:AddLabel("Use WASD to move. Speed multiplies your normal WalkSpeed.")

-- Additional help tab (demonstration of another tab)
local InfoTab = Window:AddTab("Help")
InfoTab:AddLabel("1. Enable CFrame Speed")
InfoTab:AddLabel("2. Adjust multiplier (1x-50x)")
InfoTab:AddLabel("3. Move using WASD")
InfoTab:AddLabel("Note: This directly moves your character's CFrame, may cause clipping or anti-cheat detection.")

-- Startup notification (appears after 1 second)
task.delay(1, function()
    Window:Notify("CFrame Speed Loaded", "RightShift to toggle UI | Use Speed tab", 4)
end)
