-- Загрузка библиотеки
local MinecraftLib = loadstring(game:HttpGet("https://raw.githubusercontent.com/Mamaksimaaa/Roblox-script/refs/heads/main/Lib/load.lua"))()

-- Создание окна
local Window = MinecraftLib:CreateWindow("Minecraft UI v3", {
    Theme = "Overworld" -- Overworld / Nether / End
})

-- ==================== ВКЛАДКА 1: Main ====================
local MainTab = Window:AddTab("Main")

MainTab:AddLabel("Welcome to Minecraft UI!")

MainTab:AddButton("Click Me!", function()
    Window:Notify("Button", "You clicked the button!", 3)
end)

MainTab:AddSeparator("Features")

local autoFarmToggle = MainTab:AddToggle("Auto Farm", false, function(state)
    Window:Notify("Toggle", "Auto Farm: " .. tostring(state), 2)
    print("[Auto Farm] State:", state)
end)

local speedSlider = MainTab:AddSlider("Walk Speed", 16, 200, 50, function(val)
    print("[Speed] Value:", val)
    -- game.Players.LocalPlayer.Character.Humanoid.WalkSpeed = val
end)

MainTab:AddTextbox("Player Name", "Enter name...", function(text)
    Window:Notify("Textbox", "Entered: " .. text, 2)
    print("[Player] Name:", text)
end)

local gameModeDropdown = MainTab:AddDropdown("Game Mode", {"Survival", "Creative", "Adventure", "Spectator"}, function(selected)
    Window:Notify("Dropdown", "Selected: " .. selected, 2)
    print("[Mode] Selected:", selected)
end)

local toggleUIKeybind = MainTab:AddKeybind("Toggle UI", Enum.KeyCode.F, function()
    Window:ToggleVisible()
end)

local accentColor = MainTab:AddColorPicker("Accent Color", Color3.fromRGB(80, 140, 45), function(color)
    print("[Color] Selected:", color)
end)

-- ==================== ВКЛАДКА 2: Player ====================
local PlayerTab = Window:AddTab("Player")

PlayerTab:AddLabel("Player Settings")

PlayerTab:AddButton("Heal to Full", function()
    Window:Notify("Heal", "Healed to full HP!", 2)
    print("[Heal] Full health restored")
end)

PlayerTab:AddButton("Suicide", function()
    Window:Notify("Death", "You died!", 2)
    print("[Suicide] Player died")
end)

PlayerTab:AddSeparator("Stats")

local healthSlider = PlayerTab:AddSlider("Max Health", 100, 1000, 100, function(val)
    print("[Health] Max:", val)
end)

local jumpSlider = PlayerTab:AddSlider("Jump Power", 50, 300, 100, function(val)
    print("[Jump] Power:", val)
    -- game.Players.LocalPlayer.Character.Humanoid.JumpPower = val
end)

local noclipToggle = PlayerTab:AddToggle("Noclip", false, function(state)
    print("[Noclip] State:", state)
end)

local godModeToggle = PlayerTab:AddToggle("God Mode", false, function(state)
    print("[God] State:", state)
end)

PlayerTab:AddTextbox("Teleport to Player", "Username...", function(name)
    print("[Teleport] Target:", name)
    Window:Notify("Teleport", "Teleporting to " .. name .. "...", 2)
end)

-- ==================== ВКЛАДКА 3: Combat ====================
local CombatTab = Window:AddTab("Combat")

CombatTab:AddLabel("Combat Features")

    print("[Aimbot] State:", state)

    print("[Damage] Multiplier:", val)
end)

local reachSlider = CombatTab:AddSlider("Reach", 5, 50, 10, function(val)
    print("[Reach] Distance:", val)
end)

local hitboxDropdown = CombatTab:AddDropdown("Hitbox Expand", {"Head", "Torso", "All", "None"}, function(selected)
    print("[Hitbox] Mode:", selected)
end)

local autoClickToggle = CombatTab:AddToggle("Auto Clicker", false, function(state)
    print("[AutoClick] State:", state)
end)

CombatTab:AddKeybind("Aim Key", Enum.KeyCode.Q, function()
    print("[Aim] Key pressed")
end)

-- ==================== ВКЛАДКА 4: Settings ====================
local SettingsTab = Window:AddTab("Settings")

SettingsTab:AddLabel("UI Configuration")

    print("[FPS] Cap:", val)
end)

local fovSlider = SettingsTab:AddSlider("FOV", 30, 120, 70, function(val)
    print("[FOV] Value:", val)
    -- workspace.CurrentCamera.FieldOfView = val
end)

local antiBanToggle = SettingsTab:AddToggle("Anti-Ban", false, function(state)

-- Секция внутри Settings

debugSection:AddToggle("Console Logs", true, function(state)
    print("[Debug] Logs:", state)
end)

debugSection:AddToggle("Error Notifications", false, function(state)
    print("[Debug] Errors:", state)
end)

debugSection:AddButton("Dump Config", function()
    Window:Notify("Debug", "Config dumped to console!", 2)
    print("=== CONFIG DUMP ===")
    print("Auto Farm:", autoFarmToggle:Get())
    print("Walk Speed:", speedSlider:Get())
    print("Game Mode:", gameModeDropdown:Get())
    print("Accent Color:", accentColor:Get())
end)
