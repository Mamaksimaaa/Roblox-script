-- ============================================================
-- ШАГ 1: СОЗДАНИЕ ОКНА  (MinecraftLib:CreateWindow)
-- ============================================================
--
--   Сигнатура:
--       MinecraftLib:CreateWindow(title, config)
--       MinecraftLib:CreateWindow(config)   -- можно без отдельного title
--
--   Псевдонимы:
--       MinecraftLib.Window(...)
--       MinecraftLib.New(...)
--
--   Параметры config:
--       Title        (string)  — заголовок окна
--       Theme        (string)  — тема по умолчанию: "Overworld"|"Nether"|"End"|"Sea"|"Sky"
--       ToggleKey    (string|EnumItem) — клавиша показа/скрытия; по умолчанию RightShift
--       Size         ({X=, Y=}) — начальный размер окна в пикселях
--       ConfigFolder (string)  — папка для сохранения конфигов (по умолчанию — название окна)
--       AutoSave     (bool)    — автоматически сохранять конфиг при любом изменении
--       AutoLoad     (bool)    — загружать autosave при старте (работает вместе с AutoSave)
--       Scale        (number)  — начальный масштаб UI (0.5 – 1.5)

local MinecraftLib = loadstring(game:HttpGet("https://raw.githubusercontent.com/Mamaksimaaa/Roblox-script/refs/heads/main/Lib/load.lua"))()

local Window = MinecraftLib:CreateWindow({
    Title       = "My Minecraft UI",
    Theme       = "Overworld",   -- стартовая тема
    ToggleKey   = "RightShift",  -- Enum.KeyCode.RightShift
    Size        = {X = 620, Y = 460},
    ConfigFolder = "MyScript",
    AutoSave    = true,
    AutoLoad    = true,
    Scale       = 1.0,
})

-- После создания окно сразу видно на экране.
-- На мобильных устройствах автоматически появляется плавающая кнопка "M".
-- В заголовке есть: иконка блока травы, название, кнопка темы (OW/NT/ED/SE/SK),
-- кнопка свернуть (–), кнопка закрыть (×).
-- Двойной клик по заголовку — максимизировать/восстановить.
-- Правый нижний угол — ресайз-хэндл.


-- ============================================================
-- ШАГ 2: МЕТОДЫ ОКНА
-- ============================================================

-- Скрыть окно (анимация сжатия)
Window:Hide()

-- Показать окно (анимация разворачивания)
Window:Show()

-- Переключить видимость
Window:ToggleVisible()

-- Изменить заголовок в шапке окна
Window:SetTitle("Новый заголовок")

-- Изменить клавишу переключения (принимает строку или EnumItem)
Window:SetToggleKey("RightAlt")

-- Изменить масштаб UI (0.5 = 50%, 1.5 = 150%)
Window:SetScale(1.1)

-- Сменить тему
--   Доступные встроенные темы: "Overworld", "Nether", "End", "Sea", "Sky"
Window:SetTheme("Nether")

-- Получить текущую тему
local theme, themeName = Window:GetTheme()
-- theme    — таблица с цветами (Color3 значения)
-- themeName — строка, например "Nether"

-- Получить список всех тем
local allThemes = Window:GetThemes()
-- вернёт {"Overworld", "Nether", "End", "Sea", "Sky", ...ваши кастомные...}

-- Добавить кастомную тему
--   Первый аргумент — название, второй — таблица переопределений.
--   Все недостающие поля берутся из темы "Overworld".
MinecraftLib:AddTheme("MyTheme", {
    Background    = Color3.fromRGB(20, 20, 30),
    Accent        = Color3.fromRGB(100, 200, 255),
    AccentHover   = Color3.fromRGB(130, 220, 255),
    AccentPress   = Color3.fromRGB(70, 160, 220),
    TextPrimary   = Color3.fromRGB(255, 255, 255),
    TextSecondary = Color3.fromRGB(180, 200, 220),
    -- остальные поля = Overworld
})
Window:SetTheme("MyTheme")


-- ============================================================
-- ШАГ 3: ВОДЯНОЙ ЗНАК (SetWatermark)
-- ============================================================
--
--   Показывает плавающий HUD в верхнем левом углу экрана.
--   Отображает: название скрипта | FPS | Ping (мс).
--   Водяной знак можно перетаскивать мышью.
--
--   Window:SetWatermark(enabled, text)
--       enabled (bool)   — включить/выключить
--       text    (string) — текст перед FPS/Ping (необязательно)

Window:SetWatermark(true, "MyScript v1.0")
-- Теперь показывает: "MyScript v1.0  |  120 FPS  |  32 ms"

Window:SetWatermark(false) -- скрыть водяной знак


-- ============================================================
-- ШАГ 4: УВЕДОМЛЕНИЯ (Notify)
-- ============================================================
--
--   Появляются в правом нижнем углу экрана, стекируются (до 6 штук),
--   содержат прогресс-бар и кнопку закрытия (клик по уведомлению).
--
--   Сигнатура 1 (короткая):
--       Window:Notify(title, message, duration, type)
--
--   Сигнатура 2 (таблица):
--       Window:Notify({
--           Title    = "...",
--           Content  = "...",   -- (псевдоним: Message, Text)
--           Duration = 5,       -- секунд (по умолчанию 4)
--           Type     = "...",   -- "Success" | "Warning" | "Error" | nil (accent-цвет)
--       })
--
--   Псевдоним: Window.Notification(...)
--
--   Возвращает: { Dismiss = function }

local notif = Window:Notify("Заголовок", "Текст уведомления", 4)
notif.Dismiss() -- закрыть вручную

Window:Notify({Title="Успех!", Content="Конфиг сохранён.", Type="Success"})
Window:Notify({Title="Внимание", Content="Включите функцию X.", Type="Warning", Duration=6})
Window:Notify({Title="Ошибка", Content="Не удалось загрузить.", Type="Error"})

-- ШАГ 6: ВКЛАДКИ (AddTab)
-- ============================================================
--
--   Вкладки — вертикальный список кнопок слева.
--   Первая добавленная вкладка открывается автоматически.
--
--   Window:AddTab(name)           — строка
--   Window:AddTab({Name=..., Icon=...}) — таблица (Icon — rbxassetid://)
--
--   Псевдонимы: Window.Tab, Window.CreateTab
--
--   Методы объекта Tab:
--       Tab:Select()        — программно выбрать вкладку
--       Tab.Name            — название вкладки
--       Tab:AddNotify(...)  — псевдоним Window:Notify
--
--   Все Add-методы доступны напрямую через Tab (AddButton, AddToggle и т.д.)
--   а также через псевдонимы: Tab.Button, Tab.Toggle, Tab.Slider,
--   Tab.Textbox, Tab.Input, Tab.Dropdown, Tab.Label, Tab.Paragraph,
--   Tab.Keybind, Tab.ColorPicker, Tab.Section, Tab.Separator

local MainTab     = Window:AddTab("Main")
local SettingsTab = Window:AddTab({Name = "Visual", Icon = "rbxassetid://123456"})
local PlayerTab   = Window:AddTab("Player")

-- ============================================================
-- ШАГ 7: СЕКЦИИ (AddSection)
-- ============================================================
--
--   Складная группа элементов внутри вкладки.
--   Клик по заголовку секции сворачивает/разворачивает содержимое.
--
--   Tab:AddSection(title, defaultOpen)
--       title       (string) — заголовок
--       defaultOpen (bool)   — открыта по умолчанию (true)
--
--   Возвращает объект Section, который имеет те же Add-методы, что и Tab.
--   Методы секции:
--       Section:SetOpen(bool)   — свернуть/развернуть
--       Section:Destroy()       — удалить секцию и её содержимое

local VisualSection  = MainTab:AddSection("Visuals", false) -- закрыта

-- Добавить элемент внутрь секции — то же самое, что в Tab:
-- CombatSection:AddToggle(...)
-- CombatSection:AddSlider(...)
-- и т.д.

-- Закрыть секцию из кода:
MiscSection:SetOpen(false)

-- ШАГ 8: КНОПКА (AddButton)
-- ============================================================
--
--   Сигнатура 1 (позиционная):
--       Tab:AddButton(text, callback)
--
--   Сигнатура 2 (таблица):

--           Name     = "...",
--           Callback = function() end,

-- ============================================================
-- ШАГ 9: ПЕРЕКЛЮЧАТЕЛЬ (AddToggle)
-- ============================================================
--
--   Сигнатура 1 (позиционная):
--       Tab:AddToggle(text, default, callback)
--
--   Сигнатура 2 (таблица):
--       Tab:AddToggle({
--           Name     = "...",
--           Default  = false,
--           Callback = function(value) end,  -- value: bool
--           Flag     = "...",    -- уникальный ключ для системы конфигов
--           Keybind  = "...",    -- клавиша быстрого переключения (строка/EnumItem)
--           Tooltip  = "...",
--       })
--
--   Возвращает объект с методами:
--       el:Set(bool, silent)    — установить значение (silent=true → без Callback)
--       el:Get()                — вернуть текущее bool значение
--       el:Toggle()             — инвертировать
--       el:SetText(string)      — изменить подпись
--       el:SetKeybind(key)      — изменить горячую клавишу
--       + стандартные SetVisible/SetEnabled/IsEnabled/Destroy

local AimbotToggle = CombatSection:AddToggle({
    Name    = "Aimbot",
    Default = false,
    Flag    = "aimbot_enabled",   -- используется при сохранении конфига
    Keybind = "H",                -- Toggle по нажатию H
    Tooltip = "Автоматическое прицеливание",
    Callback = function(value)
        if value then
            print("Aimbot включён")
        else
            print("Aimbot выключен")
        end
    end,
})

-- Включить программно (без вызова Callback):
AimbotToggle:Set(true, true)

-- Включить и вызвать Callback:
AimbotToggle:Set(true)

-- Переключить:
AimbotToggle:Toggle()

-- Прочитать состояние:
local isOn = AimbotToggle:Get()
print("Aimbot state:", isOn)

-- ШАГ 10: СЛАЙДЕР (AddSlider)
-- ============================================================
--
--   Сигнатура 1 (позиционная):
--       Tab:AddSlider(text, min, max, default, callback)
--
--   Сигнатура 2 (таблица):
--       Tab:AddSlider({
--           Name     = "...",
--           Min      = 0,
--           Max      = 100,
--           Default  = 50,
--           Step     = 1,       -- шаг (поддерживает дробные: 0.1, 0.5 и т.д.)
--           Suffix   = "",      -- строка после числа, например " m" или "%"
--           Callback = function(value) end,  -- value: number
--           Flag     = "...",
--           Tooltip  = "...",
--       })
--
--   Значение можно ввести вручную, кликнув на число справа.
--
--   Возвращает объект с методами:
--       el:Set(number, silent)        — установить значение
--       el:Get()                      — вернуть текущее число

local SpeedSlider = PlayerTab:AddSlider({
    Name     = "Walk Speed",
    Min      = 16,
    Max      = 500,
    Default  = 16,
    Step     = 1,
    Suffix   = " studs/s",
    Flag     = "walk_speed",
    Tooltip  = "Скорость ходьбы персонажа",

        local char = game.Players.LocalPlayer.Character
        if char and char:FindFirstChild("Humanoid") then
            char.Humanoid.WalkSpeed = value
        end
    end,
})

-- Маленький шаг (дробный):

-- ============================================================
-- ШАГ 13: ГОРЯЧАЯ КЛАВИША (AddKeybind)
-- ============================================================
--
--   Отображает кнопку с текущей клавишей, клик → режим ожидания ("..."),
--   затем нажмите любую клавишу — она запишется.
--   Escape = отменить. Backspace/Delete = очистить (None).
--
--   Сигнатура (таблица):
--       Tab:AddKeybind({
--           Name      = "...",
--           Default   = "F",             -- строка или EnumItem
--           Callback  = function(pressed) end,  -- pressed: bool (Press-режим не передаёт)
--           OnChanged = function(key) end,      -- key: EnumItem или nil
--           Mode      = "Press",         -- "Press" — один раз при нажатии
--                                        -- "Hold"  — Callback(true) при зажатии, Callback(false) при отпускании
--           Flag      = "...",
--           Tooltip   = "...",
--       })
--
--   Возвращает объект с методами:
--       el:Get()                — вернуть EnumItem или nil
--       el:Set(key, silent)     — задать клавишу
--       el:SetCallback(fn)      — изменить обработчик
--       + стандартные SetVisible/SetEnabled/IsEnabled/Destroy

local SprintKey = PlayerTab:AddKeybind({
    Name    = "Sprint Key",
    Default = "LeftShift",
    Mode    = "Hold",
    Flag    = "sprint_key",
    Callback = function(held)
        local char = game.Players.LocalPlayer.Character
        if char and char:FindFirstChild("Humanoid") then
            char.Humanoid.WalkSpeed = held and 32 or 16
        end
    end,
    OnChanged = function(key)
        print("Новая клавиша спринта:", key and key.Name or "None")
    end,
})

local PanicKey = MiscSection:AddKeybind({
    Name    = "Panic / Hide UI",
    Default = "Delete",
    Mode    = "Press",
    Flag    = "panic_key",
    Callback = function()
        Window:ToggleVisible()
    end,
})

-- Прочитать текущую клавишу:
local key = SprintKey:Get()
if key then print(key.Name) end

-- ШАГ 14: ЦВЕТОВОЙ ПИКЕР (AddColorPicker)
-- ============================================================
--
--   Открывается всплывающим окном при клике на цветной квадрат.
--   Содержит: SV-поле, полосу оттенка (Hue), опциональный Alpha-канал,
--   HEX-поле для ввода, RGB-метка, 9 цветовых пресетов.
--
--   Сигнатура (таблица):
--       Tab:AddColorPicker({
--           Name               = "...",
--           Default            = Color3.fromRGB(255, 0, 0),
--           Alpha              = false,            -- включить канал прозрачности
--           DefaultTransparency = 0,               -- начальное значение прозрачности (0=непрозрачный)
--           Callback           = function(color, alpha) end,
--               -- color: Color3
--               -- alpha: number (0–1) — только если Alpha=true
--           Flag    = "...",
--           Tooltip = "...",

--       el:SetTransparency(val, silent) — задать transparency
--       el:Close()                      — закрыть пикер
--       + стандартные SetVisible/SetEnabled/IsEnabled/Destroy

local ChamsColor = VisualSection:AddColorPicker({

    Flag    = "chams_color",

        -- Применить ко всем NPC:
        for _, model in ipairs(workspace:GetDescendants()) do
            if model:IsA("BasePart") and model.Parent:FindFirstChild("Humanoid") then
                model.Color = color
            end
        end
    end,
})
