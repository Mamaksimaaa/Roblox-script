-- ============================================================
-- ШАГ 1: ЗАГРУЗКА И СОЗДАНИЕ ОКНА
-- ============================================================
--
--   MinecraftLib:CreateWindow(config)
--   Псевдонимы: MinecraftLib.Window(...)  /  MinecraftLib.New(...)
--
--   Параметры config:
--     Title        (string)          — заголовок в шапке окна
--     Theme        (string)          — стартовая тема:
--                                      "Overworld" | "Nether" | "End" | "Sea" | "Sky"
--     ToggleKey    (string|EnumItem) — клавиша показа/скрытия (по умолч. RightShift)
--     Size         ({X=, Y=})        — начальный размер окна в пикселях
--     ConfigFolder (string)          — имя подпапки для конфигов
--     AutoSave     (bool)            — сохранять конфиг автоматически при изменениях
--     AutoLoad     (bool)            — загружать autosave при старте
--     Scale        (number)          — начальный масштаб UI (0.5 – 1.5)

local MinecraftLib = loadstring(game:HttpGet(
    "https://raw.githubusercontent.com/Mamaksimaaa/Roblox-script/refs/heads/main/Lib/load.lua"
))()

local Window = MinecraftLib:CreateWindow({
    Title        = "My Minecraft UI",
    Theme        = "Overworld",
    ToggleKey    = "RightShift",
    Size         = {X = 620, Y = 460},
    ConfigFolder = "MyScript",
    AutoSave     = true,
    AutoLoad     = true,
    Scale        = 1.0,
})

-- После создания окно сразу видно.
-- На мобильных автоматически появляется плавающая кнопка "M".
-- Шапка содержит: иконка блока, заголовок, кнопка темы (OW/NT/ED/SE/SK),
-- кнопка настроек (⚙), кнопка свернуть (–), кнопка закрыть (×).
-- Двойной клик по заголовку — максимизировать/восстановить.
-- Правый нижний угол — хэндл изменения размера.


-- ============================================================
-- ШАГ 2: МЕТОДЫ ОКНА
-- ============================================================

Window:Hide()           -- скрыть (анимация сжатия)
Window:Show()           -- показать (анимация разворачивания)
Window:ToggleVisible()  -- переключить видимость

Window:SetTitle("Новый заголовок")   -- изменить заголовок
Window:SetToggleKey("RightAlt")      -- строка или EnumItem
Window:SetScale(1.1)                 -- масштаб UI, 0.5–1.5

-- Сменить тему (встроенные: Overworld, Nether, End, Sea, Sky)
Window:SetTheme("Nether")

-- Получить текущую тему
local theme, themeName = Window:GetTheme()
-- theme     — таблица Color3-полей текущей темы
-- themeName — строка, например "Nether"

-- Получить все доступные темы
local allThemes = Window:GetThemes()
-- → {"Overworld","Nether","End","Sea","Sky", ...кастомные...}

-- Добавить кастомную тему (вызов на MinecraftLib, не на Window)
-- Недостающие поля наследуются из "Overworld"
MinecraftLib:AddTheme("MyTheme", {
    Background    = Color3.fromRGB(20, 20, 30),
    Accent        = Color3.fromRGB(100, 200, 255),
    AccentHover   = Color3.fromRGB(130, 220, 255),
    AccentPress   = Color3.fromRGB(70,  160, 220),
    TextPrimary   = Color3.fromRGB(255, 255, 255),
    TextSecondary = Color3.fromRGB(180, 200, 220),
})
Window:SetTheme("MyTheme")

-- Встроенная боковая панель настроек (⚙ в шапке)
Window:OpenSettings()   -- открыть
Window:CloseSettings()  -- закрыть
Window:ToggleSettings() -- переключить

-- Кастомный фоновый рисунок (URL изображения)
Window:SetCustomBackground("https://i.imgur.com/example.png")

-- Переключить вкладки в горизонтальный режим (сверху)
Window:SetTabMode(true)   -- горизонталь
Window:SetTabMode(false)  -- вертикаль (по умолчанию)
local isHoriz = Window:IsHorizontalTabs()  -- → bool


-- ============================================================
-- ШАГ 3: ВОДЯНОЙ ЗНАК (SetWatermark)
-- ============================================================
--
--   Плавающий HUD в верхнем левом углу.
--   Показывает: <текст>  |  <FPS> FPS  |  <Ping> ms
--   Можно перетаскивать мышью.
--
--   Window:SetWatermark(enabled, text?)
--     enabled (bool)   — включить/выключить
--     text    (string) — текст (необязательно)

Window:SetWatermark(true, "MyScript v1.0")
-- → "MyScript v1.0  |  120 FPS  |  32 ms"

Window:SetWatermark(false)  -- скрыть


-- ============================================================
-- ШАГ 4: УВЕДОМЛЕНИЯ (Notify)
-- ============================================================
--
--   Появляются в правом нижнем углу, стекируются (до 6 штук).
--   Содержат прогресс-бар. Клик или таймер — закрытие.
--
--   Сигнатура 1 (позиционная):
--     Window:Notify(title, message, duration, type)
--
--   Сигнатура 2 (таблица):
--     Window:Notify({
--         Title    = "...",
--         Content  = "...",    -- псевдонимы: Message, Text
--         Duration = 5,        -- секунд (по умолчанию 4)
--         Type     = "...",    -- "Success" | "Warning" | "Error" | nil
--     })
--
--   Псевдоним: Window.Notification(...)
--   Возвращает: { Dismiss = function }

local notif = Window:Notify("Заголовок", "Текст", 4)
notif.Dismiss()  -- закрыть программно

Window:Notify({Title = "Успех!",    Content = "Конфиг сохранён.",      Type = "Success"})
Window:Notify({Title = "Внимание",  Content = "Включите функцию X.",   Type = "Warning", Duration = 6})
Window:Notify({Title = "Ошибка",    Content = "Не удалось загрузить.", Type = "Error"})

-- ШАГ 13: ГОРЯЧАЯ КЛАВИША (AddKeybind)
-- ============================================================
--
--   Клик на кнопку → режим ожидания ("...") → нажмите клавишу.
--   Escape — отменить. Backspace/Delete — очистить (None).
--
--   Tab:AddKeybind({
--       Name=, Default="F",
--       Mode="Press",    — "Press" (один раз) | "Hold" (зажатие/отпускание)
--       Callback=fn(),   — Press: без аргументов
--                          Hold:  fn(true) при зажатии, fn(false) при отпускании
--       OnChanged=fn(key), — key: EnumItem или nil
--       Flag=, Tooltip=
--   })
--
--   Псевдоним: Tab.Keybind(...)
--
--   Возвращает объект:
--     el:Get() → EnumItem или nil
--     el:Set(key, silent?)
--     el:SetCallback(fn)
--     + SetVisible/SetEnabled/IsEnabled/Destroy

local SprintKey = PlayerTab:AddKeybind({
    Name    = "Sprint",
    Default = "LeftShift",
    Mode    = "Hold",
    Flag    = "sprint_key",
    Callback = function(held)
        local char = game.Players.LocalPlayer.Character
        if char and char:FindFirstChild("Humanoid") then
            char.Humanoid.WalkSpeed = held and 32 or 16
        end
    end,
    OnChanged = function(key)
        print("Клавиша спринта:", key and key.Name or "None")
    end,
})

local PanicKey = MiscSec:AddKeybind({
    Name    = "Panic / Hide UI",
    Default = "Delete",
    Mode    = "Press",
    Flag    = "panic_key",
    Callback = function()
        Window:ToggleVisible()
    end,
})

local curKey = SprintKey:Get()
if curKey then print(curKey.Name) end

-- ШАГ 19: СИСТЕМА КОНФИГОВ
-- ============================================================
--
--   Конфиги хранятся в:
--       MinecraftUI/configs/<ConfigFolder>/<name>.json
--
--   Требуется файловая система (isfolder, writefile и т.д.).
--   Элементы с Flag= сохраняются/восстанавливаются автоматически.

-- Сохранить
local ok, err = Window:SaveConfig("default")
if ok then print("Сохранено!") else print("Ошибка:", err) end

-- Загрузить
local ok2, err2 = Window:LoadConfig("default")
if not ok2 then print("Загрузка не удалась:", err2) end

-- Список конфигов
local configs = Window:GetConfigs()  -- → {string,...}
for _, name in ipairs(configs) do print("Конфиг:", name) end

-- Удалить
local deleted = Window:DeleteConfig("old_config")  -- → bool

-- Загрузить autosave
Window:LoadAutoSave()
-- Автосохранение срабатывает через 1 с после любого изменения
-- (если AutoSave=true при CreateWindow)

--   Это позволяет SaveConfig/LoadConfig сохранять его состояние.

-- Доступ к элементу по флагу:
local el = Window._flags["speed_value"]
if el then el:Set(80) end


-- ============================================================
-- ШАГ 21: СОБЫТИЯ ТЕМЫ (RegisterThemed)
-- ============================================================
--
--   Регистрирует функцию, которая вызывается при каждой смене темы.
--   Используется для перекраски кастомных элементов.
--
--   Window:RegisterThemed(fn, watchInstance?)
--     fn            — function(theme) — таблица Color3-полей
--     watchInstance — Instance; при его уничтожении колбэк снимается

local myFrame = Instance.new("Frame")
myFrame.Parent = game.Players.LocalPlayer.PlayerGui

Window:RegisterThemed(function(t)
    myFrame.BackgroundColor3 = t.Accent
    -- Доступные поля темы:
    -- Background, SecondaryBG, TertiaryBG, Glass
    -- Accent, AccentHover, AccentPress
    -- TabActive, TabInactive
    -- TextPrimary, TextSecondary, TextDisabled, TextHover
    -- Border, BevelLight, BevelDark, Shadow
    -- SliderFill, SliderBG, ToggleOn, ToggleOff
    -- Scrollbar, Title, CloseBtn
    -- Notification, NotifAccent, Separator
    -- BackgroundImage, TitleBarImage
    -- Gradient (ColorSequence — только Sea/Sky)
    -- GradientRotation (number)
end, myFrame)


-- ============================================================
-- ШАГ 22: УНИЧТОЖЕНИЕ ОКНА (Destroy)
-- ============================================================
--
--   Полностью удаляет UI и отключает все соединения
--   (InputBegan, InputEnded, Heartbeat, слежение за камерой).
--   После вызова Window использовать нельзя.

task.delay(120, function()
    Window:Destroy()
    print("UI уничтожен")
end)

-- ШАГ 23: ПОЛНЫЙ РАБОЧИЙ ПРИМЕР

do
    local Lib = loadstring(game:HttpGet(
        "https://raw.githubusercontent.com/Mamaksimaaa/Roblox-script/refs/heads/main/Lib/load.lua"
    ))()

    local W = Lib:CreateWindow({
        Title        = "Example Script",
        Theme        = "Sea",
        ToggleKey    = "RightControl",
        ConfigFolder = "ExampleScript",
        AutoSave     = true,
        AutoLoad     = true,
    })

    W:SetWatermark(true, "Example v1.0")

    -- ── Combat ──────────────────────────────────────────────
    local combat    = W:AddTab("Combat")
    local combatSec = combat:AddSection("Settings")

    local aimbotToggle = combatSec:AddToggle({
        Name = "Aimbot", Default = false, Flag = "aimbot",
        Callback = function(v) print("Aimbot:", v) end,
    })
    local smoothSlider = combatSec:AddSlider({
        Name = "Smooth", Min = 1, Max = 20, Default = 5, Step = 1, Flag = "smooth",
        Callback = function(v) print("Smooth:", v) end,
    })
    local fovSlider = combatSec:AddSlider({
        Name = "FOV Circle", Min = 30, Max = 360, Default = 90, Suffix = "°", Flag = "fov",
        Callback = function(v) print("FOV:", v) end,
    })
    combatSec:AddButton({
        Name = "Reset to defaults",
        Callback = function()
            aimbotToggle:Set(false)
            smoothSlider:Set(5)
            fovSlider:Set(90)
            W:Notify("Combat", "Настройки сброшены.", 3, "Warning")
        end,
    })

    -- ── Visuals ─────────────────────────────────────────────
    local visuals = W:AddTab("Visuals")
    local espSec  = visuals:AddSection("ESP")

    espSec:AddToggle({
        Name = "Players ESP", Default = false, Flag = "esp_players",
        Callback = function(v) print("ESP:", v) end,
    })
    espSec:AddColorPicker({
        Name = "Box Color", Default = Color3.fromRGB(0, 200, 255), Flag = "esp_col",
        Callback = function(c) print("Color:", c) end,
    })
    espSec:AddSlider({
        Name = "Box Thickness", Min = 1, Max = 5, Default = 2, Step = 1, Flag = "esp_thick",
        Callback = function(v) print("Thickness:", v) end,
    })

    -- ── Player ──────────────────────────────────────────────
    local player = W:AddTab("Player")

    player:AddSlider({
        Name = "WalkSpeed", Min = 16, Max = 500, Default = 16, Step = 1, Flag = "wsp",
        Callback = function(v)
            local c = game.Players.LocalPlayer.Character
            if c and c:FindFirstChild("Humanoid") then c.Humanoid.WalkSpeed = v end
        end,
    })
    player:AddSlider({
        Name = "JumpPower", Min = 50, Max = 500, Default = 50, Step = 5, Flag = "jp",
        Callback = function(v)
            local c = game.Players.LocalPlayer.Character
            if c and c:FindFirstChild("Humanoid") then c.Humanoid.JumpPower = v end
        end,
    })
    player:AddDropdown({
        Name = "Team", Options = {"Red", "Blue", "Green"}, Flag = "team",
        Callback = function(v) print("Team:", v) end,
    })

    -- ── Misc ────────────────────────────────────────────────
    local misc = W:AddTab("Misc")

    misc:AddKeybind({
        Name = "Panic key", Default = "End", Mode = "Press", Flag = "panic",
        Callback = function() W:ToggleVisible() end,
    })
    misc:AddTextbox({
        Name = "Discord Webhook", Placeholder = "https://discord.com/api/...", Flag = "webhook",
        Callback = function(v) print("Webhook:", v) end,
    })
    misc:AddParagraph("Credits", "MinecraftLib v3.5\nАвтор: YourName")
    misc:AddButton({
        Name = "Notify Test",
        Callback = function()
            W:Notify({Title = "Test", Content = "Тестовое уведомление!", Type = "Success"})
        end,
    })
    misc:AddButton({
        Name = "Dialog Test",
        Callback = function()
            W:Dialog({
                Title   = "Подтверждение",
                Content = "Сбросить все настройки?",
                Buttons = {
                    {Text = "Да",  Callback = function() print("Сброс!") end},
                    {Text = "Нет"},
                },
            })
        end,
    })

    -- Встроенная вкладка Settings (тема, масштаб, конфиги, watermark)
    W:AddSettingsTab("Settings")

    print("Example загружен! RightControl — показать/скрыть.")
end

-- СПРАВОЧНИК ВСЕХ МЕТОДОВ
-- ============================================================
--
--  БИБЛИОТЕКА:
--    MinecraftLib:CreateWindow(config)  → Window
--    MinecraftLib:AddTheme(name, tbl)   → theme
--    MinecraftLib.Window(...)  /  MinecraftLib.New(...)
--
--  ОКНО:
--    :AddTab(name/config)               → Tab
--    :SelectTab(name)                   → bool
--    :AddSettingsTab(name?)             → Tab
--    :SetTheme(name)
--    :GetTheme()                        → theme, name
--    :GetThemes()                       → {string,...}
--    :SetTitle(string)
--    :SetToggleKey(key)
--    :SetScale(0.5–1.5)
--    :SetWatermark(bool, text?)
--    :SetCustomBackground(url)
--    :SetTabMode(bool)
--    :IsHorizontalTabs()                → bool
--    :Show() / :Hide() / :ToggleVisible()
--    :OpenSettings() / :CloseSettings() / :ToggleSettings()
--    :Notify(title, msg, dur, type)     → {Dismiss}
--    :Notify({Title,Content,Duration,Type})
--    :Dialog({Title,Content,Buttons})   → {Close}
--    :SaveConfig(name)                  → ok, err
--    :LoadConfig(name)                  → ok, err
--    :GetConfigs()                      → {string,...}
--    :DeleteConfig(name)                → bool
--    :LoadAutoSave()                    → ok, err
--    :RegisterThemed(fn, instance?)
--    :RegisterFlag(flag, element)
--    :Destroy()
--
--  ВКЛАДКА / СЕКЦИЯ (одинаковые методы):
--    :AddButton(...)       → el {SetText, SetCallback}
--    :AddToggle(...)       → el {Set, Get, Toggle, SetText, SetKeybind}
--    :AddSlider(...)       → el {Set, Get, SetRange}
--    :AddTextbox(...)      → el {Set, Get, .Box}
--    :AddDropdown(...)     → el {Set, Get, Refresh, Close, GetOptions}
--    :AddKeybind(...)      → el {Set, Get, SetCallback}
--    :AddColorPicker(...)  → el {Set, Get, GetHex, GetTransparency, SetTransparency, Close}
--    :AddLabel(...)        → el {Set, Get, .Label}
--    :AddParagraph(...)    → el {Set, SetContent}
--    :AddSeparator(text?)  → el
--    :AddSection(...)      → Section (те же методы + SetOpen, Destroy)
--    :AddNotify(...)       — псевдоним Window:Notify
--    :Select()             — (только Tab) выбрать программно
--
--  ПСЕВДОНИМЫ ADD-МЕТОДОВ (Tab и Section):
--    .Button = AddButton       .Toggle  = AddToggle
--    .Slider = AddSlider       .Textbox = AddTextbox
--    .Input  = AddTextbox      .Dropdown = AddDropdown
--    .Label  = AddLabel        .Paragraph = AddParagraph
--    .Keybind = AddKeybind     .ColorPicker = AddColorPicker
--    .Section = AddSection     .Separator = AddSeparator
--
--  ОБЩИЕ МЕТОДЫ ВСЕХ ЭЛЕМЕНТОВ:
--    :SetVisible(bool)
--    :SetEnabled(bool)  — затемнить/активировать
--    :IsEnabled()       → bool
--    :Destroy()
--
--  ПОЛЯ ТАБЛИЦЫ ТЕМЫ (Color3, если не указано иное):
--    Background, SecondaryBG, TertiaryBG, Glass
--    Accent, AccentHover, AccentPress
--    TabActive, TabInactive
--    TextPrimary, TextSecondary, TextDisabled, TextHover
--    Border, BevelLight, BevelDark, Shadow
--    SliderFill, SliderBG, ToggleOn, ToggleOff
--    Scrollbar, Title, CloseBtn
--    Notification, NotifAccent, Separator
--    BackgroundImage (string или false)
--    TitleBarImage   (string или false)
--    Gradient        (ColorSequence — Sea/Sky)
--    GradientRotation (number)
