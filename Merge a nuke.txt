-- ─── Загрузка библиотеки MinecraftLib ────────────────────────────────────
local Lib = loadstring(game:HttpGet("https://raw.githubusercontent.com/Mamaksimaaa/Roblox-script/refs/heads/main/Minecraft%20library/Minecraftlibrary.lua"))()

-- ─── REMOTES ─────────────────────────────────────────────────────────────
local PurchaseUpgrade = game:GetService("ReplicatedStorage")
    :WaitForChild("NukeRemotes")
    :WaitForChild("PurchaseUpgrade")

local RequestLockBase = game:GetService("ReplicatedStorage")
    :WaitForChild("NukeRemotes")
    :WaitForChild("RequestLockBase")

-- ─── СОСТОЯНИЕ ────────────────────────────────────────────────────────────
local State = {
    running             = false,
    thread              = nil,
    interval            = 100,          -- мс
    doLockbase          = true,
    doTier              = true,
    doMax               = true,
    doRequestLockBase   = true,
    totalFires          = 0,
}

-- ─── ЛОГИКА АПГРЕЙДОВ ────────────────────────────────────────────────────
local function fireUpgrades()
    if State.doLockbase then
        PurchaseUpgrade:FireServer("LOCKBASE")
        State.totalFires = State.totalFires + 1
    end
    if State.doTier then
        PurchaseUpgrade:FireServer("TIER")
        State.totalFires = State.totalFires + 1
    end
    if State.doMax then
        PurchaseUpgrade:FireServer("MAX")
        State.totalFires = State.totalFires + 1
    end
    if State.doRequestLockBase then
        RequestLockBase:FireServer()
        State.totalFires = State.totalFires + 1
    end
end

local statusLabel  -- будет заполнено после создания UI

local function runLoop()
    while State.running do
        fireUpgrades()
        if statusLabel then
            statusLabel:Set("Всего отправлено: " .. State.totalFires)
        end
        task.wait(State.interval / 1000)
    end
end

local function startLoop()
    if State.running then return end
    if not (State.doLockbase or State.doTier or State.doMax or State.doRequestLockBase) then
        Window:Notify("Ошибка", "Выбери хотя бы один апгрейд.", 3)
        return
    end
    State.running = true
    State.thread = task.spawn(runLoop)
end

local function stopLoop()
    State.running = false
    if State.thread then
        task.cancel(State.thread)
        State.thread = nil
    end
end

-- ─── СКРЫТИЕ NotifFrame ─────────────────────────────────────────────────
local function hideNotifFrames()
    local playerGui = game:GetService("Players").LocalPlayer
        :WaitForChild("PlayerGui")

    for _, obj in ipairs(playerGui:GetDescendants()) do
        if obj:IsA("Frame") and obj.Name:lower():find("notif") then
            obj.Visible = false
        end
    end

    playerGui.DescendantAdded:Connect(function(obj)
        if obj:IsA("Frame") and obj.Name:lower():find("notif") then
            task.defer(function()
                obj.Visible = false
            end)
        end
    end)
end

task.spawn(hideNotifFrames)

-- ─── СОЗДАНИЕ ОКНА ───────────────────────────────────────────────────────
local Window = Lib:CreateWindow("Nuke Auto Upgrade", { Theme = "Overworld" })

-- ─── ВКЛАДКИ ─────────────────────────────────────────────────────────────
local MainTab = Window:AddTab("Main")
local SettTab = Window:AddTab("Settings")

-- ════════════════════════════════════════════════════════════
--  MAIN TAB
-- ════════════════════════════════════════════════════════════

-- Раздел: цели апгрейда
MainTab:AddLabel("═══ Upgrade Targets ═══")  -- заголовок секции

local lockbaseToggle = MainTab:AddToggle("LOCKBASE", true, function(v)
    State.doLockbase = v
end)

MainTab:AddLabel("")  -- отступ

local tierToggle = MainTab:AddToggle("TIER", true, function(v)
    State.doTier = v
end)

MainTab:AddLabel("")

local maxToggle = MainTab:AddToggle("MAX", true, function(v)
    State.doMax = v
end)

MainTab:AddLabel("")

local reqLockToggle = MainTab:AddToggle("RequestLockBase", true, function(v)
    State.doRequestLockBase = v
end)

MainTab:AddSeparator()  -- разделитель

-- Раздел: управление
MainTab:AddLabel("═══ Control ═══")

local startStopBtn  -- сохраним для изменения текста
startStopBtn = MainTab:AddButton("▶  Start Auto Upgrade", function()
    if State.running then
        stopLoop()
        startStopBtn.Text = "▶  Start Auto Upgrade"
        Window:Notify("Остановлено", "Цикл апгрейдов остановлен.", 2)
    else
        startLoop()
        if State.running then
            startStopBtn.Text = "⏹  Stop Auto Upgrade"
            Window:Notify("Запущено", "Цикл апгрейдов активен!", 2)
        end
    end
end)

MainTab:AddLabel("")

MainTab:AddButton("Однократный запуск", function()
    local before = State.totalFires
    fireUpgrades()
    local fired = State.totalFires - before
    Window:Notify("Выполнено", "Отправлено " .. fired .. " запрос(ов).", 2)
    if statusLabel then
        statusLabel:Set("Всего отправлено: " .. State.totalFires)
    end
end)

MainTab:AddSeparator()

-- Раздел: статус
MainTab:AddLabel("═══ Status ═══")

statusLabel = MainTab:AddLabel("Ожидание запуска...")
-- обновим начальное значение
statusLabel:Set("Всего отправлено: 0")

MainTab:AddLabel("NotifFrame скрывается автоматически ✓")

-- ════════════════════════════════════════════════════════════
--  SETTINGS TAB
-- ════════════════════════════════════════════════════════════

-- Раздел: интервал
SettTab:AddLabel("═══ Interval ═══")

local intervalSlider = SettTab:AddSlider("Delay (мс)", 50, 2000, 100, function(v)
    State.interval = v
end)

SettTab:AddSeparator()

-- Раздел: уведомления
SettTab:AddLabel("═══ Notifications ═══")

SettTab:AddToggle("Скрывать NotifFrame", true, function(v)
    if v then
        task.spawn(hideNotifFrames)
    end
end)

SettTab:AddLabel("")

SettTab:AddButton("Скрыть NotifFrame сейчас", function()
    local count = 0
    local playerGui = game:GetService("Players").LocalPlayer:WaitForChild("PlayerGui")
    for _, obj in ipairs(playerGui:GetDescendants()) do
        if obj:IsA("Frame") and obj.Name:lower():find("notif") then
            obj.Visible = false
            count = count + 1
        end
    end
    Window:Notify("NotifFrame", "Скрыто фреймов: " .. count, 3)
end)

SettTab:AddSeparator()

-- Раздел: прочее
SettTab:AddLabel("═══ Misc ═══")

SettTab:AddButton("Сбросить счётчик", function()
    State.totalFires = 0
    if statusLabel then
        statusLabel:Set("Всего отправлено: 0")
    end
    Window:Notify("Сброшено", "Счётчик обнулён.", 2)
end)

-- ─── СТАРТОВОЕ УВЕДОМЛЕНИЕ ──────────────────────────────────────────────
task.delay(1, function()
    Window:Notify("IRY HUB", "MERGE A NUKE", 4)
end)
