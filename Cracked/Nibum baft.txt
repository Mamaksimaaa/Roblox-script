local function createFrame(parent, size, position, color, cornerRadius, zIndex)
    local frame = Instance.new("Frame", parent)
    frame.Size = size
    frame.Position = position
    frame.BackgroundColor3 = color or Colors.bg1
    frame.BorderSizePixel = 0
    if cornerRadius then
        local corner = Instance.new("UICorner", frame)
        corner.CornerRadius = UDim.new(0, cornerRadius)
    end
    if zIndex then
        frame.ZIndex = zIndex
    end
    return frame
end

local function createTextLabel(parent, text, textSize, color, font, alignment)
    local label = Instance.new("TextLabel", parent)
    label.Size = UDim2.new(1, 0, 1, 0)
    label.Position = UDim2.new(0, 0, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextSize = textSize or 13
    label.TextColor3 = color or Colors.txt
    label.Font = font or Enum.Font.GothamMedium
    label.TextXAlignment = alignment or Enum.TextXAlignment.Left
    label.TextYAlignment = Enum.TextYAlignment.Center
    label.TextWrapped = false
    return label
end

local function createTextButton(parent, size, position, color, cornerRadius, zIndex)
    local btn = Instance.new("TextButton", parent)
    btn.Size = size
    btn.Position = position
    btn.BackgroundColor3 = color or Colors.bg3
    btn.BorderSizePixel = 0

        corner.CornerRadius = UDim.new(0, cornerRadius)
    end
    if zIndex then

    return btn
end

local function addStroke(object, color, thickness, transparency)
    local stroke = Instance.new("UIStroke", object)
    stroke.Color = color or Colors.brd
    stroke.Thickness = thickness or 1
    stroke.Transparency = transparency or 0
    return stroke
end

local function tween(object, duration, properties, style, direction)
    local info = TweenInfo.new(duration, style or Enum.EasingStyle.Linear, direction or Enum.EasingDirection.InOut)
    local tween = TweenService:Create(object, info, properties)
    tween:Play()
    return tween
end

-- Основная загрузка скрипта
local function loadMainScript()
    setStatus("Connecting to server...", Colors.gold)
    task.wait(0.5)
    setStatus("Downloading script...", Colors.acc)
    task.wait(0.3)

    -- НОВАЯ ССЫЛКА (buildaboatfortreasure)
    local scriptUrl = "https://raw.githubusercontent.com/privnerxsc-unknown/unknownyea/refs/heads/main/buildaboatfortreasure"
    local success, result = pcall(function()
        return game:HttpGet(scriptUrl)
    end)

    if success and result then
        setStatus("Executing script...", Colors.grn)
        task.wait(0.2)
        local loadFunction = loadstring(result)
        if loadFunction then
            setStatus("Success! Loading...", Colors.grn)
            task.wait(0.5)
            loadFunction()
        else
            setStatus("Error: Failed to compile script!", Colors.acc)
            task.wait(2)
        end
    else
        setStatus("Error: Could not download script!", Colors.acc)
        task.wait(2)
    end
    closeUI()
end
